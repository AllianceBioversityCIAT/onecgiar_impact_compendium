# !/usr/bin/env bash
# File: scripts/start_local.sh
# Purpose: local build + run of Impact Compendium (frontend + backend)
# Author: Amazon Q Automation

set -e
set -o pipefail

echo "🚀 Starting Impact Compendium Local Environment"

# Check Node
command -v node >/dev/null || { echo "❌ Node.js missing"; exit 1; }

# Check Python
command -v python3 >/dev/null || { echo "❌ Python3 missing"; exit 1; }

echo "✅ Dependencies found"

# Load environment variables
export DB_HOST=db-impact-compendium.c9s2e8cg81v6.us-east-2.rds.amazonaws.com
export DB_PORT=3306
export DB_NAME=impact
export DB_USER=root
export DB_PASS=pas
export API_PORT=8000
export VITE_API_BASE_URL=http://localhost:${API_PORT}

# Create venv if needed
cd Backend
if [ ! -d \".venv\" ]; then
  python3 -m venv .venv
fi
source .venv/bin/activate
pip install -r requirements.txt

# Simple DB check
python -c \"import pymysql; conn=pymysql.connect(host='${DB_HOST}',user='${DB_USER}',password='${DB_PASS}',database='${DB_NAME}'); cur=conn.cursor(); cur.execute('SELECT 1'); print('✅ DB connection OK'); conn.close()\"

# Run backend
echo \"▶️ Starting FastAPI on http://localhost:${API_PORT}\"
(uvicorn app.main:app --reload --port ${API_PORT} &)
BACK_PID=$!

# Run frontend
cd ../Frontend
echo \"▶️ Starting React (Vite) on http://localhost:5173\"
npm install
npm run dev &
FRONT_PID=$!

trap \"echo '🛑 Stopping servers...'; kill $BACK_PID $FRONT_PID\" INT

wait
